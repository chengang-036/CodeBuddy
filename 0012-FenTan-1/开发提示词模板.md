```
【标准处理流程】
1. 创建新workbook：XLSX.utils.book_new()

3. 转换为二维数组：XLSX.utils.sheet_to_json(sheet, {header: 1, defval: ''})

5. 转换回sheet：XLSX.utils.aoa_to_sheet(array)

【进度反馈】
- 将计算拆分为5-6个步骤
- 每个步骤更新进度条（10%, 20%, 40%, 60%, 80%, 95%, 100%）
- 使用async/await + sleep()让UI有时间更新
- 显示当前步骤的文字说明
```

### 1. 空值处理

```
// 严格的空值检查
if (value !== '' && value !== null && value !== undefined) {
```

### 2. 百分比处理

```
【百分比格式化】
function formatPercentage(value) {
    if (value === '' || value === null || value === undefined || isNaN(value)) {
        return '';
    }
    return (value * 100).toFixed(2) + '%';
}

【百分比解析】
function parsePercentage(str) {
    if (str === '' || str === null || str === undefined) return 0;
    if (typeof str === 'number') return str;
    if (typeof str === 'string' && str.includes('%')) {
        return parseFloat(str.replace('%', '')) / 100;
    }
    return parseFloat(str) || 0;
}

【使用场景】
- 存储时使用小数形式（0.1234）
- 显示时转换为百分比字符串（12.34%）
- 计算时先解析为小数
```

### 3. 分类聚合计算

```
【按条件聚合】
// 示例：计算所有管理岗的总和
let totalManagement = 0;
for (let i = 1; i < sheet.length; i++) {
    const attr = String(sheet[i][attrCol] || '').trim();
    if (attr === '管理岗') {
        const value = parsePercentage(sheet[i][valueCol]);
        totalManagement += value;
    }
}
```

### 4. 按比例分摊计算

```
【分摊计算模式】
// 模式1：按数量比例分摊
const ratio = individualCount / totalCount;
const allocated = totalAmount * ratio;
const newValue = currentValue + allocated;

// 模式2：按映射比例分摊
const ratio = mappingTable[category];  // 从配置表查找比例
const allocated = totalAmount * ratio;
const newValue = currentValue + allocated;

【分摊流程】
1. 先复制原值到目标列
2. 计算被分摊方的总额
3. 计算分摊基数（总人力、总数量等）
4. 遍历接收方，按比例分摊
5. 清空被分摊方的目标列
```

### 5. 合计行计算

```
【添加合计行】
const totalRow = new Array(headers.length).fill('');
totalRow[0] = '合计';  // 第一列显示"合计"

// 数值列：直接求和
if (numCol !== -1) {
    let total = 0;
    for (let i = 1; i < sheet.length; i++) {
        const value = parseFloat(sheet[i][numCol]);
        if (!isNaN(value)) total += value;
    }
    totalRow[numCol] = total;
}

// 百分比列：解析后求和再格式化
if (percentCol !== -1) {
    let total = 0;
    for (let i = 1; i < sheet.length; i++) {
        total += parsePercentage(sheet[i][percentCol]);
    }
    totalRow[percentCol] = formatPercentage(total);
}

sheet.push(totalRow);
```

### 文件读取模板

```javascript
function handleFile(file) {
    if (!file.name.endsWith('.xlsx')) {
        showStatus('error', '请上传 .xlsx 格式的Excel文件！');
        return;
    }

    if (typeof XLSX === 'undefined') {
        alert('错误：Excel处理库未加载\n\n请检查：\n1. 电脑是否联网\n2. 按Ctrl+F5刷新页面\n3. 或联系技术支持');
        return;
    }

    fileName = file.name;
    showStatus('info', '正在读取文件...');

    const reader = new FileReader();
    reader.onload = (e) => {
        try {
            const data = new Uint8Array(e.target.result);
            workbook = XLSX.read(data, { type: 'array' });
            validateSheets();
        } catch (error) {
            showStatus('error', '文件读取失败：' + error.message);
        }
    };
    reader.onerror = () => {
        showStatus('error', '文件读取失败，请重试。');
    };
    reader.readAsArrayBuffer(file);
}
```

### 计算流程模板

```javascript
async function performCalculations() {
    processedWorkbook = XLSX.utils.book_new();
    
    // 复制所有sheet
    workbook.SheetNames.forEach(name => {
        const ws = XLSX.utils.sheet_to_json(workbook.Sheets[name], { header: 1, defval: '' });
        processedWorkbook.Sheets[name] = XLSX.utils.aoa_to_sheet(ws);
        processedWorkbook.SheetNames.push(name);
    });

    updateProgress(10, '步骤1/5：读取数据...');
    await sleep(100);

    // 转换为二维数组
    const sheet1 = XLSX.utils.sheet_to_json(workbook.Sheets['1'], { header: 1 });
    const sheet2 = XLSX.utils.sheet_to_json(workbook.Sheets['2'], { header: 1 });
    const sheet3 = XLSX.utils.sheet_to_json(workbook.Sheets['3'], { header: 1, defval: '' });

    updateProgress(20, '步骤2/5：[步骤描述]...');
    await sleep(100);
    step2Function(sheet1, sheet3);

    updateProgress(40, '步骤3/5：[步骤描述]...');
    await sleep(100);
    step3Function(sheet3);

    // ... 更多步骤

    updateProgress(95, '保存结果...');
    processedWorkbook.Sheets['3'] = XLSX.utils.aoa_to_sheet(sheet3);
    updateProgress(100, '计算完成！');
}
```
